-- Q1. Top 5 highest-value orders with customer names

SELECT c.CustomerName,
       o.OrderID,
       o.OrderAmount,
       o.Status
FROM Orders o
JOIN Customers c ON c.CustomerID = o.CustomerID
ORDER BY o.OrderAmount DESC
LIMIT 5;
-- Q2. Customers whose names start with A, R, or S

SELECT CustomerID,
       CustomerName,
       City,
       JoinDate
FROM Customers
WHERE CustomerName REGEXP '^[ARS]';
-- Q3. Orders greater than ₹10,000 with Pending status

SELECT c.CustomerName,
       o.OrderID,
       o.OrderAmount,
       o.Status
FROM Orders o
JOIN Customers c ON c.CustomerID = o.CustomerID
WHERE o.OrderAmount > 10000
  AND o.Status = 'Pending';
-- Q4. Customers associated for more than 365 days

SELECT CustomerID,
       CustomerName,
       JoinDate,
       DATEDIFF(CURDATE(), JoinDate) AS Days_Associated
FROM Customers
WHERE DATEDIFF(CURDATE(), JoinDate) > 365
ORDER BY Days_Associated DESC;
-- Q5. GST and Expected Collection Date

SELECT OrderID,
       OrderAmount AS Current_Amount,
       ROUND(OrderAmount * 0.18, 2) AS GST_18_Percent,
       ROUND(OrderAmount * 1.18, 2) AS Total_Amount_After_GST,
       DATE_ADD(OrderDate, INTERVAL 15 DAY) AS Expected_Collection_Date
FROM Orders
ORDER BY OrderID;
-- Q6. Orders ranked 11th to 20th

SELECT o.OrderID,
       c.CustomerName,
       o.OrderAmount,
       o.Status
FROM Orders o
JOIN Customers c
    ON c.CustomerID = o.CustomerID
ORDER BY o.OrderAmount DESC
LIMIT 10 OFFSET 10;
-- Q7. Customers who never placed an order

SELECT c.CustomerID,
       c.CustomerName,
       c.JoinDate,
       DATEDIFF(CURDATE(), c.JoinDate) AS Days_Since_Registration
FROM Customers c
LEFT JOIN Orders o
    ON c.CustomerID = o.CustomerID
WHERE o.CustomerID IS NULL
ORDER BY c.CustomerID;
-- Q8. Employee hierarchy using JOIN

SELECT 
    e.EmployeeName AS Employee_Name,
    COALESCE(m.EmployeeName, 'No Manager') AS Manager_Name,
    e.Department
FROM Employees e
LEFT JOIN Employees m
    ON e.ManagerID = m.EmployeeID
ORDER BY e.EmployeeID;
-- Q9. Customers whose first order was within 30 days of joining

SELECT 
    c.CustomerName,
    c.JoinDate,
    MIN(o.OrderDate) AS First_Order_Date,
    DATEDIFF(MIN(o.OrderDate), c.JoinDate) AS Days_Taken_To_Place_First_Order
FROM Customers c
JOIN Orders o
    ON c.CustomerID = o.CustomerID
GROUP BY c.CustomerID, c.CustomerName, c.JoinDate
HAVING DATEDIFF(MIN(o.OrderDate), c.JoinDate) BETWEEN 0 AND 30
ORDER BY c.CustomerID;
-- Q10. Customer Purchase and Loyalty Report

SELECT c.CustomerName,
       COALESCE(SUM(o.OrderAmount), 0) AS Total_Purchase_Amount,
       CASE
           WHEN COALESCE(SUM(o.OrderAmount), 0) > 50000 THEN 'Premium'
           WHEN COALESCE(SUM(o.OrderAmount), 0) BETWEEN 20000 AND 50000 THEN 'Gold'
           ELSE 'Silver'
       END AS Spending_Category,
       CASE
           WHEN c.JoinDate < '2024-01-01' THEN 'Loyal Customer'
           ELSE 'New Customer'
       END AS Loyalty_Status
FROM Customers c
LEFT JOIN Orders o
    ON o.CustomerID = c.CustomerID
GROUP BY c.CustomerID, c.CustomerName, c.JoinDate
ORDER BY Total_Purchase_Amount DESC;
-- Q11. Orders higher than average order amount

SELECT c.CustomerName,
       o.OrderID,
       o.OrderAmount
FROM Orders o
JOIN Customers c
    ON c.CustomerID = o.CustomerID
WHERE o.OrderAmount > (SELECT AVG(OrderAmount) FROM Orders)
ORDER BY o.OrderAmount DESC;
-- Q12. 2nd to 6th highest orders

SELECT c.CustomerName,
       o.OrderID,
       o.OrderAmount
FROM Orders o
JOIN Customers c
    ON c.CustomerID = o.CustomerID
ORDER BY o.OrderAmount DESC
LIMIT 5 OFFSET 1;
